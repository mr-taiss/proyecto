// Reglas de PRUEBA aislada. Conservar firebase_diagnostico y objetos_prueba.
// Administradores adicionales: solo los puede autorizar un administrador autenticado.
rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  function signed() { return request.auth != null; }
  function hasProfile() { return signed() && exists(/databases/$(database)/documents/perfiles_prueba/$(request.auth.uid)); }
  function role() { return get(/databases/$(database)/documents/perfiles_prueba/$(request.auth.uid)).data.rol; }
  function isAdmin() { return hasProfile() && role() == 'admin' && get(/databases/$(database)/documents/perfiles_prueba/$(request.auth.uid)).data.get('activo', true) == true; }
  function student() { return hasProfile() && role() == 'estudiante'; }
  function activeProfile() { return hasProfile() && get(/databases/$(database)/documents/perfiles_prueba/$(request.auth.uid)).data.get('activo', true) == true && (role() == 'admin' || (role() == 'estudiante' && get(/databases/$(database)/documents/perfiles_prueba/$(request.auth.uid)).data.get('requiereCambioClave', false) != true)); }
  match /firebase_diagnostico/{uid} { allow read,write: if signed() && request.auth.uid == uid; }
  match /objetos_prueba/{uid}/items/{id} {
   function owner(){return signed() && request.auth.uid == uid;}
   function valid(){return request.resource.data.keys().hasOnly(['nombre','lugar','estado','creado']) && request.resource.data.keys().hasAll(['nombre','lugar','estado','creado']) && request.resource.data.nombre is string && request.resource.data.nombre.size()>0 && request.resource.data.nombre.size()<=80 && request.resource.data.lugar is string && request.resource.data.lugar.size()>0 && request.resource.data.lugar.size()<=80 && request.resource.data.estado in ['Perdido','Encontrado','Recuperado'];}
   allow get,list: if owner();
   allow create: if owner() && valid() && request.resource.data.creado == request.time;
   allow update: if owner() && valid() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['estado']) && request.resource.data.creado == resource.data.creado;
   allow delete: if owner();
  }
  match /perfiles_prueba/{uid} {
   allow get: if signed() && (request.auth.uid == uid || activeProfile());
   allow list: if isAdmin();
   allow delete: if false;
   // Retirar acceso conserva la cuenta y sus registros. No permite retirarse a uno mismo.
   allow update: if isAdmin() && request.auth.uid != uid && resource.data.rol == 'admin'
     && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['activo'])
     && request.resource.data.activo == false;
   allow update: if signed() && request.auth.uid == uid
     && resource.data.rol == 'estudiante'
     && resource.data.get('requiereCambioClave', false) == true
     && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['requiereCambioClave'])
     && request.resource.data.requiereCambioClave == false;
   allow create: if isAdmin()
     && request.resource.data.keys().hasOnly(['alias','curso','rol'])
     && request.resource.data.keys().hasAll(['alias','curso','rol'])
     && request.resource.data.rol == 'admin'
     && request.resource.data.curso == ''
     && request.resource.data.alias is string
     && request.resource.data.alias.size() >= 3
     && request.resource.data.alias.size() <= 80;
   // Las cuentas estudiantiles ya están asignadas. El registro libre queda cerrado.
  }
  match /objetos_compartidos_prueba/{itemId} {
   function valid(){return request.resource.data.keys().hasOnly(['nombre','lugar','estado','creadorUid','creado','categoria','descripcion','fechaEvento','curso','imageData','registranteNombre']) && request.resource.data.keys().hasAll(['nombre','lugar','estado','creadorUid','creado','categoria','descripcion','fechaEvento','curso']) && request.resource.data.nombre is string && request.resource.data.nombre.size()>0 && request.resource.data.nombre.size()<=80 && request.resource.data.lugar is string && request.resource.data.lugar.size()>0 && request.resource.data.lugar.size()<=80 && request.resource.data.estado in ['Perdido','Encontrado','Recuperado'] && request.resource.data.categoria is string && request.resource.data.categoria.size()>0 && request.resource.data.categoria.size()<=80 && request.resource.data.descripcion is string && request.resource.data.descripcion.size()>0 && request.resource.data.descripcion.size()<=500 && request.resource.data.fechaEvento is string && request.resource.data.fechaEvento.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$') && request.resource.data.curso is string && request.resource.data.curso.size()<=12 && (!('imageData' in request.resource.data) || (request.resource.data.imageData is string && request.resource.data.imageData.size() <= 200000 && request.resource.data.imageData.matches('^data:image/jpeg;base64,[A-Za-z0-9+/=]+$'))) && (!('registranteNombre' in request.resource.data) || (request.resource.data.registranteNombre is string && request.resource.data.registranteNombre.size() <= 100)) && (isAdmin() ? request.resource.data.curso == '' : request.resource.data.curso == get(/databases/$(database)/documents/perfiles_prueba/$(request.auth.uid)).data.curso) && request.resource.data.creadorUid == request.auth.uid && request.resource.data.creado == request.time;}
   // El usuario autorizó expresamente la consulta pública de los registros completos de los tres estados.
   // El cliente público limita la consulta a estos estados.
   allow get,list: if activeProfile() || resource.data.estado in ['Perdido','Encontrado','Recuperado'];
   allow create: if activeProfile() && (isAdmin() || student()) && valid() && (isAdmin() || request.resource.data.estado == 'Perdido');
   allow update: if isAdmin() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['estado']) && request.resource.data.estado in ['Perdido','Encontrado','Recuperado'];
   allow delete: if isAdmin();
  }
  match /{document=**} {allow read,write: if false;}
 }
}
